cmake_minimum_required(VERSION 3.16)
set(PROJECT_VER "0.1.0")
if(NOT XIAOTAI_BOARD_DIR)
    message(FATAL_ERROR "Select an isolated XiaoTai profile with python scripts/build.py")
endif()
set(COMPONENTS main)
include(cmake/select_board.cmake)
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
set(ENV{ESP_IDF_VERSION} "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
idf_build_set_property(DEPENDENCIES_LOCK "${DEPENDENCIES_LOCK}")
idf_build_set_property(XIAOTAI_BOARD_DIR "${XIAOTAI_BOARD_DIR}")
idf_build_set_property(XIAOTAI_PROFILE_ID "${XIAOTAI_PROFILE_ID}")
idf_build_set_property(XIAOTAI_NATIVE_PROFILE "${XIAOTAI_NATIVE_PROFILE}")
project(xiaotai)
if(TARGET lvgl_port_lib)
    include(cmake/lvgl_idf6_dsi.cmake)
endif()
if(IDF_TARGET STREQUAL "esp32s3")
    include(cmake/lvgl_psram_pool.cmake)
    starter_enable_lvgl_psram_pool()
    include(cmake/tflm_conv_channels.cmake)
    starter_enable_tflm_conv_channels()
endif()
if(XIAOTAI_PROFILE_ID STREQUAL "cores3")
    include("${XIAOTAI_SELECTED_BOARD}/build-support/cmake/tflm_dilated_depthwise.cmake")
    starter_enable_tflm_dilated_depthwise()
    if(CONFIG_XIAOTAI_MQTT_PSRAM_BUFFERS)
        include("${XIAOTAI_SELECTED_BOARD}/build-support/cmake/mqtt_psram_buffers.cmake")
        starter_enable_mqtt_psram_buffers()
    endif()
endif()
if(IDF_TARGET STREQUAL "esp32p4")
    include(cmake/p4_board_contract.cmake)
endif()
if(XIAOTAI_PROFILE_ID STREQUAL "xmini")
    include(cmake/xmini_usb_monitor.cmake)
endif()
