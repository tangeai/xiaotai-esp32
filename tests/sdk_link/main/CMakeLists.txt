idf_component_register(SRCS "link_check.c" INCLUDE_DIRS "." REQUIRES tirtc_sdk)
if(CMAKE_BUILD_EARLY_EXPANSION)
    return()
endif()
# Whole-archive alone still permits --gc-sections to discard every SDK function.
get_target_property(sdk_library tirtc_prebuilt IMPORTED_LOCATION)
execute_process(COMMAND "${CMAKE_NM}" --defined-only --extern-only "${sdk_library}"
    OUTPUT_VARIABLE symbols COMMAND_ERROR_IS_FATAL ANY)
string(REGEX MATCHALL "[^\r\n]+" symbol_lines "${symbols}")
set(roots "")
foreach(line IN LISTS symbol_lines)
    if(line MATCHES "^[0-9a-fA-F]+ [TW] ([A-Za-z_][A-Za-z_0-9]*)$")
        list(APPEND roots "${CMAKE_MATCH_1}")
    endif()
endforeach()
list(REMOVE_DUPLICATES roots)
list(LENGTH roots root_count)
if(root_count LESS 1000)
    message(FATAL_ERROR "SDK symbol extraction failed: only ${root_count} functions")
endif()
foreach(symbol IN LISTS roots)
    target_link_libraries(${COMPONENT_LIB} INTERFACE "-Wl,-u,${symbol}")
endforeach()
string(REPLACE ";" "\n" root_text "${roots}")
file(WRITE "${CMAKE_BINARY_DIR}/sdk-link-roots.txt" "${root_text}\n")
