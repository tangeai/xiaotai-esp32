# Host build of the SDK and the applications: every component compiles without
# ESP-IDF against the host port, and each component's own tests (its test/
# directory) run under the sanitizers. This file never names a component:
# components are found on disk and pull in their dependencies themselves
# (sdk/cmake/xiaotai_host.cmake), so adding one needs no edit here.
#
#   -DXIAOTAI_COMPONENTS="a;b"            only these components and what they depend on
#   -DXIAOTAI_BOARD=<vendor>/<board>      also the board package boards/<vendor>/<board> (component
#                                         xiaotai_board); alone, only the board and what it depends on
#   -DXIAOTAI_HOST_DEPENDENCY_TESTS=OFF   skip the tests of components pulled in as dependencies
#   -DXIAOTAI_ROOT=<dir>                  search another tree for components (helper self-test)
#
# tests/sdk/run.py wraps configure, build and CTest.
cmake_minimum_required(VERSION 3.16)
project(xiaotai_sdk_host C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

option(XIAOTAI_HOST_TESTS "Build and register each component's test/ directory" ON)
option(XIAOTAI_HOST_DEPENDENCY_TESTS "Also test components that are only dependencies of the selected ones" ON)
option(XIAOTAI_HOST_ALLOW_MISSING "Report tests skipped for missing pinned sources" OFF)
set(XIAOTAI_COMPONENTS "" CACHE STRING "Components to build and test; empty selects every component found")
set(XIAOTAI_BOARD "" CACHE STRING "Board package to build and test, as <vendor>/<board> under boards/")

include("${CMAKE_CURRENT_LIST_DIR}/../../sdk/cmake/xiaotai_host.cmake")
enable_testing()
file(REMOVE "${CMAKE_BINARY_DIR}/host-skipped.txt")
file(REMOVE "${CMAKE_BINARY_DIR}/host-stack-tests.txt")
add_compile_options(${XIAOTAI_HOST_COMPILE_OPTIONS})
add_link_options(${XIAOTAI_HOST_LINK_OPTIONS})

if(XIAOTAI_COMPONENTS)
    set(XIAOTAI_HOST_SELECTED ${XIAOTAI_COMPONENTS})
elseif(XIAOTAI_BOARD)
    set(XIAOTAI_HOST_SELECTED "")
else()
    xiaotai_host_components(XIAOTAI_HOST_SELECTED)
endif()
if(XIAOTAI_BOARD)
    list(APPEND XIAOTAI_HOST_SELECTED xiaotai_board)
endif()
if(NOT XIAOTAI_HOST_SELECTED)
    message(FATAL_ERROR "No component found under ${XIAOTAI_ROOT}")
endif()
xiaotai_host_require(${XIAOTAI_HOST_SELECTED})
if(NOT XIAOTAI_HOST_SANITIZE)
    return()
endif()

get_filename_component(source_root "${CMAKE_CURRENT_LIST_DIR}/../.." REALPATH)
if(XIAOTAI_HOST_TESTS AND XIAOTAI_ROOT STREQUAL source_root)
    add_subdirectory(integration)
endif()
if(XIAOTAI_HOST_TESTS AND NOT XIAOTAI_COMPONENTS AND NOT XIAOTAI_BOARD AND XIAOTAI_ROOT STREQUAL source_root)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME sdk_vendor_idle_stack_profile
             COMMAND Python3::Interpreter -B "${source_root}/tests/test_vendor_idle_stack_profile.py")
    add_test(NAME sdk_bench_tools COMMAND Python3::Interpreter -B -m unittest
             test_app_stacks test_boot_check test_camera_evidence test_camera_sequence
             test_device_console_aliases test_device_console_companion_sdk
             test_device_console_screen test_device_console_sdk test_flash_baud test_generate_at_reference test_partition_plan
             test_gate test_gate_ai test_gate_pair test_gate_room test_phone_ui test_sdk_soak test_snapshot
             test_wx_phone test_xt_sdk test_xtp2 test_xtp2_receive)
    set_tests_properties(sdk_bench_tools PROPERTIES LABELS "tools" TIMEOUT 120
                         WORKING_DIRECTORY "${source_root}/tools/xt")
endif()

# All-headers check over every component that was built. Each public header
# (include/**/*.h and *.hpp) must compile when it is the first include of a
# translation unit (.h as C and as C++, .hpp as C++), must be safe to include
# twice, and all of them must compile together in one C and one C++ unit. The
# units are linked into one executable, so a header that defines an object or a
# non-inline function fails here as a duplicate symbol.
function(xiaotai_write_if_changed path content)
    if(EXISTS "${path}")
        file(READ "${path}" previous)
        if(previous STREQUAL content)
            return()
        endif()
    endif()
    file(WRITE "${path}" "${content}")
endfunction()

get_property(built GLOBAL PROPERTY XIAOTAI_HOST_COMPONENTS)
set(generated "${CMAKE_BINARY_DIR}/all_headers")
set(banner "// Generated by tests/sdk/CMakeLists.txt. Do not edit.\n")
set(all_c "${banner}")
set(all_cxx "${banner}")
set(units "")
set(seen "")
set(index 0)
foreach(component IN LISTS built)
    get_property(dir GLOBAL PROPERTY "XIAOTAI_HOST_DIR_${component}")
    file(GLOB_RECURSE headers CONFIGURE_DEPENDS RELATIVE "${dir}/include"
         "${dir}/include/*.h" "${dir}/include/*.hpp")
    list(SORT headers)
    foreach(header IN LISTS headers)
        if(header IN_LIST seen)
            message(FATAL_ERROR "Public header ${header} is published by two components (second: ${component})")
        endif()
        list(APPEND seen "${header}")
        math(EXPR index "${index} + 1")
        string(MAKE_C_IDENTIFIER "${header}" id)
        set(alone "${banner}#include \"${header}\"\n#include \"${header}\"\n")
        string(APPEND all_cxx "#include \"${header}\"\n")
        xiaotai_write_if_changed("${generated}/first_${index}_${id}.cc" "${alone}")
        list(APPEND units "${generated}/first_${index}_${id}.cc")
        if(header MATCHES "\\.h$")
            string(APPEND all_c "#include \"${header}\"\n")
            xiaotai_write_if_changed("${generated}/first_${index}_${id}.c" "${alone}")
            list(APPEND units "${generated}/first_${index}_${id}.c")
        endif()
    endforeach()
endforeach()
string(APPEND all_c "int main(void)\n{\n    return 0;\n}\n")
xiaotai_write_if_changed("${generated}/all_headers.c" "${all_c}")
xiaotai_write_if_changed("${generated}/all_headers.cc" "${all_cxx}")

add_executable(sdk_all_headers "${generated}/all_headers.c" "${generated}/all_headers.cc" ${units})
target_link_libraries(sdk_all_headers PRIVATE ${built})
add_test(NAME sdk_all_headers COMMAND sdk_all_headers)
set_tests_properties(sdk_all_headers PROPERTIES TIMEOUT 120 LABELS "headers")
list(LENGTH seen header_count)
list(LENGTH built component_count)
message(STATUS "XiaoTai host build: ${component_count} components (${built}), ${header_count} public headers")
