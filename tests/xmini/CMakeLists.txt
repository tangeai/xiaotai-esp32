option(XIAOTAI_SANITIZERS "Enable address and undefined-behavior checks" OFF)
if(XIAOTAI_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
enable_testing()
get_filename_component(ROOT "../.." ABSOLUTE)
set(APP "${ROOT}/main/xiaotai/application/xmini/startup")
set(AUDIO "${ROOT}/main/xiaotai/media/xmini/xmini_audio")
set(UI "${ROOT}/main/xiaotai/ui/xmini/xmini_ui")
set(RTC "${ROOT}/main/xiaotai/transport/xmini/xmini_rtc")

set(BOARD "${ROOT}/main/boards/xmini/c3-v3/drivers/xmini_board")
set(SDK "${ROOT}/components/tirtc_sdk/variants/xmini/include")

if(NOT EXISTS "${CJSON_ROOT}/cJSON.c")
    message(FATAL_ERROR "Configure firmware dependencies first or set CJSON_ROOT")
endif()
function(xmini_test name)
    add_executable(${name} ${ARGN})
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_include_directories(${name} PRIVATE "${SDK}" "${SDK}/tirtc")
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endfunction()
