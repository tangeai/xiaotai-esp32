cmake_minimum_required(VERSION 3.16)
project(xiaotai_mz28_regression C ASM)
set(CMAKE_C_STANDARD 11)
option(XIAOTAI_SANITIZERS "Enable address and undefined-behavior checks" OFF)
if(XIAOTAI_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
enable_testing()
get_filename_component(ROOT "../.." ABSOLUTE)
set(VOIP "${ROOT}/main/xiaotai/application/mz28/wechat_voip")
set(RTC "${ROOT}/main/xiaotai/transport/mz28/rtc_adapter")
set(APP "${ROOT}/main/xiaotai/application/mz28/startup")
set(VB "${ROOT}/main/boards/mz28-ai-ppd/drivers/vb6824")
set(PLATFORM "${ROOT}/main/xiaotai/services/mz28/demo_platform")
set(CJSON_ROOT "${ROOT}/managed_components/espressif__cjson/cJSON" CACHE PATH "Resolved cJSON 1.7.19 source")
if(NOT EXISTS "${CJSON_ROOT}/cJSON.c")
    message(FATAL_ERROR "Configure firmware dependencies first or set CJSON_ROOT")
endif()
function(mz28_test name)
    add_executable(${name} "test_${name}.c" ${ARGN})
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG -ffunction-sections -fdata-sections)
    target_link_options(${name} PRIVATE -Wl,--gc-sections)
    target_include_directories(${name} PRIVATE voip_stubs stubs "${CJSON_ROOT}"
        "${VOIP}" "${VOIP}/include" "${RTC}/include" "${PLATFORM}/include"
        "${ROOT}/components/tirtc_sdk/include" "${ROOT}/components/tirtc_sdk/include/tirtc"
        "${APP}" "${VB}" "${VB}/include")
    target_link_libraries(${name} PRIVATE m)
    add_test(NAME ${name} COMMAND ${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endfunction()
mz28_test(state "${VOIP}/voip_state.c")
mz28_test(protocol "${VOIP}/voip_protocol.c" "${CJSON_ROOT}/cJSON.c")
mz28_test(boot_button_logic "${APP}/boot_button_logic.c")
mz28_test(audio audio_fixture.S "${VOIP}/sample_audio.c" "${VOIP}/opus_packet.c")
mz28_test(rtc_inputs "${RTC}/rtc_adapter.c")
mz28_test(rtc_contract "${RTC}/rtc_adapter.c")
mz28_test(voip_reject "${CJSON_ROOT}/cJSON.c")
mz28_test(voip_messages)
target_link_libraries(voip_messages PRIVATE pthread)
mz28_test(voip_join_lifetime "${VOIP}/voip_state.c" "${VOIP}/voip_protocol.c" "${CJSON_ROOT}/cJSON.c")
mz28_test(vb6824_wire "${VB}/vb6824_wire.c")
