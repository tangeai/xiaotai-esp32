cmake_minimum_required(VERSION 3.16)
project(xiaotai_lvgl_regression C CXX ASM)
get_filename_component(ROOT "../.." ABSOLUTE)
set(LVGL_ROOT "${ROOT}/managed_components/lvgl__lvgl" CACHE PATH "Resolved LVGL 9.5 source")
option(XIAOTAI_SANITIZERS "Enable ASan and UBSan" ON)
if(XIAOTAI_SANITIZERS)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
endif()
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
add_compile_definitions(LV_CONF_SKIP LV_COLOR_DEPTH=16)
add_subdirectory("${LVGL_ROOT}" lvgl EXCLUDE_FROM_ALL)
add_executable(draw_hook test_draw_hook.c "${ROOT}/components/xiaotai_lvgl/xiaotai_lvgl.c")
target_include_directories(draw_hook PRIVATE stubs "${ROOT}/components/xiaotai_lvgl/include")
target_compile_options(draw_hook PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(draw_hook PRIVATE lvgl::lvgl m)
enable_testing()
add_test(NAME draw_hook COMMAND draw_hook)
set_tests_properties(draw_hook PROPERTIES TIMEOUT 30)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/extract_key_draw.py"
    "${CMAKE_CURRENT_BINARY_DIR}/key_draw.inc" COMMAND_ERROR_IS_FATAL ANY)
add_executable(key_draw test_key_draw.c)
target_include_directories(key_draw PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_options(key_draw PRIVATE -Wall -Wextra -Werror -UNDEBUG)
target_link_libraries(key_draw PRIVATE lvgl::lvgl m)
add_test(NAME key_draw COMMAND key_draw)
set_tests_properties(key_draw PROPERTIES TIMEOUT 30)
