idf_component_register(SRCS "xmini_rtc.c" INCLUDE_DIRS "include"
                       REQUIRES tirtc_sdk freertos esp_timer)
if(CMAKE_BUILD_EARLY_EXPANSION)
    return()
endif()
idf_component_get_property(sdk_root tirtc_sdk COMPONENT_DIR)
include("${sdk_root}/select_variant.cmake")
file(READ "${sdk_dir}/contract.json" sdk_lock)
# Generic C3 link probes do not establish the Mini cancellation/diagnostic ABI.
string(JSON mini ERROR_VARIABLE mini_error GET "${sdk_lock}" xmini_contract)
if(mini_error)
    message(FATAL_ERROR "C3 SDK lacks the Xmini Mini contract: pending-close, bounded command publication and diagnostic v2. Preserve the lifecycle; rebuild the compatible producer first.")
endif()
string(JSON pending_close GET "${mini}" pending_close_contract)
string(JSON command_budget GET "${mini}" pending_command_requested_byte_budget)
if(NOT pending_close EQUAL 1 OR NOT command_budget EQUAL 16384)
    message(FATAL_ERROR "Xmini SDK pending-close or command memory contract changed")
endif()
string(JSON expected_diagnostic_header GET "${mini}" diagnostic_header_sha256)
file(SHA256 "${CMAKE_CURRENT_LIST_DIR}/include/tgtrp_internal_diag.h" actual_diagnostic_header)
if(NOT actual_diagnostic_header STREQUAL expected_diagnostic_header)
    message(FATAL_ERROR "C3 transport diagnostic header/archive mismatch")
endif()
