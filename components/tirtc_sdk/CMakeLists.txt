include("${CMAKE_CURRENT_LIST_DIR}/select_variant.cmake")
if(NOT EXISTS "${sdk_dir}/contract.json")
    message(FATAL_ERROR "No TiRTC IDF 6 SDK for ${IDF_TARGET}")
endif()
file(READ "${sdk_dir}/contract.json" contract)
string(JSON expected_hash GET "${contract}" sha256)
file(SHA256 "${sdk_dir}/libTiRTC.a" actual_hash)
if(NOT actual_hash STREQUAL expected_hash)
    message(FATAL_ERROR "TiRTC library hash does not match ${IDF_TARGET} contract")
endif()
string(JSON header_count ERROR_VARIABLE header_error LENGTH "${contract}" headers)
if(NOT header_error)
    math(EXPR header_last "${header_count}-1")
    foreach(index RANGE ${header_last})
        string(JSON header MEMBER "${contract}" headers ${index})
        string(JSON expected GET "${contract}" headers "${header}")
        file(SHA256 "${tirtc_include_dir}/tirtc/${header}" actual)
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "TiRTC header/archive identity mismatch: ${header}")
        endif()
    endforeach()
endif()
idf_component_register(SRCS "abi_contract.c" INCLUDE_DIRS "${tirtc_include_dir}"
    REQUIRES freertos esp_system esp_timer lwip mbedtls pthread WHOLE_ARCHIVE)
if(CMAKE_BUILD_EARLY_EXPANSION)
    return()
endif()
# Capability comes from the archive, not the application's TLS Kconfig. This
# says HTTPS transport exists; it does not certify the SDK's peer verification.
execute_process(COMMAND "${CMAKE_NM}" -A "${sdk_dir}/libTiRTC.a"
    OUTPUT_VARIABLE sdk_symbols COMMAND_ERROR_IS_FATAL ANY)
include("${CMAKE_CURRENT_LIST_DIR}/http_contract.cmake")
tirtc_http_capability("${contract}" "${sdk_symbols}" sdk_ssl_support)
target_compile_definitions(${COMPONENT_LIB} PUBLIC TIRTC_SDK_SSL_SUPPORT=${sdk_ssl_support})
set(idf_version "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}")
string(JSON expected_idf GET "${contract}" idf_version)
if(NOT idf_version VERSION_EQUAL expected_idf)
    message(FATAL_ERROR "TiRTC requires IDF ${expected_idf}; got ${idf_version}")
endif()
string(JSON expected_compiler GET "${contract}" compiler_version)
if(NOT CMAKE_C_COMPILER_VERSION VERSION_EQUAL expected_compiler)
    message(FATAL_ERROR "TiRTC compiler ABI mismatch")
endif()
string(JSON toolchain_release GET "${contract}" toolchain_release)
execute_process(COMMAND "${CMAKE_C_COMPILER}" --version
    OUTPUT_VARIABLE compiler_banner COMMAND_ERROR_IS_FATAL ANY)
string(FIND "${compiler_banner}" "${toolchain_release}" toolchain_match)
if(toolchain_match LESS 0)
    message(FATAL_ERROR "TiRTC toolchain release mismatch: requires ${toolchain_release}")
endif()
# SDK-owned Mbed TLS contexts are passed to the application's Mbed TLS library.
# A link succeeds even when conditional context layouts differ, so compare the
# complete crypto configuration as well as the FreeRTOS object sizes below.
include("${CMAKE_CURRENT_LIST_DIR}/crypto_contract.cmake")
tirtc_validate_crypto_contract("${contract}")
if(CONFIG_IDF_TARGET_ESP32S3)
    string(JSON libc_count ERROR_VARIABLE libc_error LENGTH "${contract}" libc_config)
    if(tirtc_board STREQUAL "lckfb/szpi-esp32s3" AND (libc_error OR NOT libc_count EQUAL 2))
        message(FATAL_ERROR "SZPI SDK requires its complete libc contract")
    endif()
    if(NOT libc_error)
        math(EXPR libc_last "${libc_count}-1")
        foreach(index RANGE ${libc_last})
            string(JSON symbol MEMBER "${contract}" libc_config ${index})
            string(JSON expected GET "${contract}" libc_config ${symbol})
            if(NOT "${${symbol}}" STREQUAL "${expected}")
                message(FATAL_ERROR "TiRTC libc ABI mismatch: ${symbol} requires ${expected}")
            endif()
        endforeach()
        execute_process(COMMAND "${CMAKE_OBJDUMP}" -d "${sdk_dir}/libTiRTC.a"
            OUTPUT_VARIABLE sdk_instructions COMMAND_ERROR_IS_FATAL ANY)
        if(CONFIG_SPIRAM AND sdk_instructions MATCHES "[ \t]s32c1i[ \t]")
            message(FATAL_ERROR "S3 PSRAM SDK embeds unsupported hardware atomics")
        endif()
    endif()
    # The S3 archive uses the consumer's PSRAM atomic policy; IDF supplies its external
    # atomic functions only with the PSRAM-aware libc implementation.
    foreach(symbol CONFIG_SPIRAM CONFIG_STDATOMIC_S32C1I_SPIRAM_WORKAROUND)
        string(JSON expected GET "${contract}" atomic_config ${symbol})
        if(NOT "${${symbol}}" STREQUAL "${expected}")
            message(FATAL_ERROR "TiRTC atomic ABI mismatch: ${symbol} requires ${expected}")
        endif()
    endforeach()
endif()
if(NOT CONFIG_FREERTOS_HZ EQUAL 1000 OR CONFIG_FREERTOS_USE_TRACE_FACILITY
   OR CONFIG_FREERTOS_USE_STATS_FORMATTING_FUNCTIONS OR CONFIG_FREERTOS_GENERATE_RUN_TIME_STATS)
    message(FATAL_ERROR "TiRTC FreeRTOS contract requires 1000 Hz, trace/stats disabled")
endif()
foreach(field IN ITEMS static_semaphore static_task tick time semaphore_alignment)
    string(JSON abi_${field} GET "${contract}" abi ${field})
    string(TOUPPER "${field}" upper)
    target_compile_definitions(${COMPONENT_LIB} PUBLIC "TIRTC_SDK_${upper}_SIZE=${abi_${field}}")
endforeach()
add_prebuilt_library(tirtc_prebuilt "${sdk_dir}/libTiRTC.a"
    REQUIRES esp_system esp_timer lwip mbedtls pthread)
target_link_libraries(${COMPONENT_LIB} INTERFACE tirtc_prebuilt)
if(CONFIG_IDF_TARGET_ESP32S3 AND CONFIG_SPIRAM)
    # Preserve the existing S3 connection-worker stack budget; C3 does not
    # inherit a PSRAM product's allocation policy.
    target_sources(${COMPONENT_LIB} PRIVATE s3_task_stack_policy.c)
    target_link_libraries(${COMPONENT_LIB} INTERFACE "-Wl,--wrap=freertos_ThreadCreateWithStackSize")
endif()
