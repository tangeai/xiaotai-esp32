cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET "esp32c3" CACHE STRING "Target chip")
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(xiaotai_xmini_c3_v3 VERSION 0.1.0)
if(NOT IDF_TARGET STREQUAL "esp32c3")
    message(FATAL_ERROR "Xmini-C3-V3 requires ESP32-C3")
endif()
include($ENV{IDF_PATH}/tools/cmake/version.cmake)
if(NOT "${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" STREQUAL "5.5.5")
    message(FATAL_ERROR "Use ESP-IDF 5.5.5 for this port")
endif()
include(cmake/mbedtls_mpi_workspace.cmake)
