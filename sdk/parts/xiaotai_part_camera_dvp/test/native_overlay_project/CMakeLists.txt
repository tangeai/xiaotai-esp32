cmake_minimum_required(VERSION 3.19)
project(dvp_overlay_probe LANGUAGES C)

set(driver "${CAMERA_DIR}/driver")
set(sccb "${driver}/sccb-ng.c")
if(mode STREQUAL "legacy_sccb")
    set(sccb "${driver}/sccb.c")
endif()
add_library(camera STATIC "${driver}/cam_hal.c" "${sccb}" "${driver}/esp_camera.c")

function(idf_component_get_property out component property)
    if(NOT component STREQUAL "espressif__esp32-camera")
        message(FATAL_ERROR "Unexpected component ${component}")
    endif()
    if(property STREQUAL "COMPONENT_DIR")
        set(${out} "${CAMERA_DIR}" PARENT_SCOPE)
    elseif(property STREQUAL "COMPONENT_LIB")
        set(${out} "camera" PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unexpected property ${property}")
    endif()
endfunction()

function(idf_build_get_property out property)
    if(NOT property STREQUAL "BUILD_COMPONENTS")
        message(FATAL_ERROR "Unexpected property ${property}")
    endif()
    set(components esp_driver_i2c)
    if(NOT mode STREQUAL "missing_camera" AND NOT mode STREQUAL "stock_missing")
        list(APPEND components espressif__esp32-camera)
    endif()
    set(${out} "${components}" PARENT_SCOPE)
endfunction()

function(idf_component_register)
    cmake_parse_arguments(PART "" "" "SRCS;INCLUDE_DIRS;PRIV_INCLUDE_DIRS;REQUIRES" ${ARGN})
    if(PART_SRCS)
        add_library(dvp STATIC ${PART_SRCS})
    else()
        add_library(dvp INTERFACE)
    endif()
    set(COMPONENT_LIB dvp PARENT_SCOPE)
endfunction()

function(idf_component_optional_requires visibility)
    if(NOT visibility STREQUAL "PRIVATE" OR NOT ARGN STREQUAL "espressif__esp32-camera;esp_driver_i2c")
        message(FATAL_ERROR "Unexpected optional prerequisites: ${visibility} ${ARGN}")
    endif()
    set_property(TARGET dvp PROPERTY fixture_prerequisites "${ARGN}")
endfunction()

set(ESP_PLATFORM TRUE)
set(XIAOTAI_PORT_V3_PRODUCT TRUE)
if(NOT mode STREQUAL "stock" AND NOT mode STREQUAL "stock_missing")
    set(CONFIG_XIAOTAI_PART_CAMERA_DVP_NATIVE TRUE)
endif()
if(mode STREQUAL "legacy_product")
    set(XIAOTAI_PORT_V3_PRODUCT FALSE)
elseif(mode STREQUAL "early_scan")
    set(CMAKE_BUILD_EARLY_EXPANSION TRUE)
endif()
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." dvp)

function(assert_selected_sources)
    get_target_property(actual camera SOURCES)
    get_target_property(adapter dvp SOURCES)
    get_target_property(prerequisites dvp fixture_prerequisites)
    get_target_property(definitions dvp COMPILE_DEFINITIONS)
    set(generated "${CMAKE_BINARY_DIR}/xiaotai_camera_dvp_overlay")
    if(mode STREQUAL "legacy_product" OR mode STREQUAL "early_scan")
        set(expected "${driver}/cam_hal.c;${driver}/sccb-ng.c;${driver}/esp_camera.c")
        if(EXISTS "${generated}" OR adapter OR prerequisites OR definitions)
            message(FATAL_ERROR "Dormant build unexpectedly activated the native adapter")
        endif()
    else()
        if(NOT adapter STREQUAL "src/camera_dvp.c;esp-idf/camera_dvp_io.c" OR NOT prerequisites)
            message(FATAL_ERROR "Missing ESP-IDF native adapter sources or prerequisites")
        endif()
        if(mode STREQUAL "stock")
            set(expected "${driver}/cam_hal.c;${driver}/sccb-ng.c;${driver}/esp_camera.c")
            if(EXISTS "${generated}" OR definitions)
                message(FATAL_ERROR "Stock DVP unexpectedly enabled the overlay")
            endif()
        else()
            set(expected "${generated}/cam_hal.c;${generated}/sccb-ng.c;${generated}/esp_camera.c")
            if(NOT "XIAOTAI_DVP_USE_OVERLAY=1" IN_LIST definitions)
                message(FATAL_ERROR "Missing explicit overlay definition")
            endif()
        endif()
    endif()
    if(NOT actual STREQUAL expected)
        message(FATAL_ERROR "Unexpected camera source selection: ${actual}")
    endif()
endfunction()
cmake_language(DEFER CALL assert_selected_sources)
