set(main_sources mqtt_client.c lib/mqtt_msg.c lib/platform_esp32_idf.c)
if(mode STREQUAL "mqtt5_sources")
    set(CONFIG_MQTT_PROTOCOL_5 TRUE)
    list(APPEND main_sources mqtt5_client.c lib/mqtt5_msg.c)
endif()
set(absolute)
foreach(source IN LISTS main_sources)
    list(APPEND absolute "${MQTT_DIR}/${source}")
endforeach()

    file(WRITE "${CMAKE_BINARY_DIR}/extra.c" "")
    list(APPEND absolute "${CMAKE_BINARY_DIR}/extra.c")
endif()
add_library(native_mqtt STATIC ${absolute})
target_include_directories(native_mqtt PUBLIC "${MQTT_DIR}/include" PRIVATE "${MQTT_DIR}/lib/include")
add_library(native_outbox STATIC "${MQTT_DIR}/lib/mqtt_outbox.c")
target_include_directories(native_outbox PUBLIC "${MQTT_DIR}/lib/mqtt_outbox/../include")
add_library(idf::mqtt::outbox ALIAS native_outbox)
add_library(native_utils STATIC "${MQTT_DIR}/lib/mqtt_utils/mqtt_utils.c")
target_include_directories(native_utils PUBLIC "${MQTT_DIR}/lib/mqtt_utils/include")
add_library(idf::mqtt::utils ALIAS native_utils)
# Only satisfy the unchanged HTTP target's registration; no HTTP logic is simulated.
file(WRITE "${CMAKE_BINARY_DIR}/esp_http_client.c" "")
add_library(native_http STATIC "${CMAKE_BINARY_DIR}/esp_http_client.c")

function(idf_component_get_property out component property)
    if(component STREQUAL "espressif__mqtt")
        if(mode STREQUAL "missing_mqtt")
            message(FATAL_ERROR "Missing locked MQTT product dependency")
        endif()
        set(directory "${MQTT_DIR}")
        set(library native_mqtt)
    elseif(component STREQUAL "esp_http_client")
        set(directory "${CMAKE_BINARY_DIR}")
        set(library native_http)
    else()
        message(FATAL_ERROR "Unexpected IDF component: ${component}")
    endif()
    if(property STREQUAL "COMPONENT_DIR")
        set(${out} "${directory}" PARENT_SCOPE)
    elseif(property STREQUAL "COMPONENT_LIB")
        set(${out} "${library}" PARENT_SCOPE)
    else()
        message(FATAL_ERROR "Unexpected component property: ${property}")
    endif()
endfunction()

function(idf_build_get_property out property)
    if(property STREQUAL "PYTHON")
        set(value "${Python3_EXECUTABLE}")
    elseif(property STREQUAL "BUILD_COMPONENTS")
        set(value espressif__mqtt)
        if(mode STREQUAL "missing_mqtt")
            set(value)
        endif()
    else()
        message(FATAL_ERROR "Unexpected build property: ${property}")
    endif()
    set(${out} "${value}" PARENT_SCOPE)
endfunction()

function(idf_component_optional_requires visibility)
    if(NOT visibility STREQUAL "PRIVATE" OR NOT ARGN STREQUAL "espressif__mqtt")
        message(FATAL_ERROR "Unexpected MQTT optional dependencies")
    endif()
    set_property(TARGET port PROPERTY fixture_mqtt_requires "${ARGN}")
endfunction()

function(idf_component_register)
    cmake_parse_arguments(PORT "" "" "SRCS;INCLUDE_DIRS;PRIV_INCLUDE_DIRS;REQUIRES;PRIV_REQUIRES" ${ARGN})
    add_library(port STATIC ${PORT_SRCS})
    set(COMPONENT_LIB port PARENT_SCOPE)
endfunction()

function(assert_mqtt_source_graph)
    get_target_property(adapter port SOURCES)
    get_target_property(actual native_mqtt SOURCES)
    if(mode STREQUAL "legacy" OR mode STREQUAL "early")
        if(EXISTS "${CMAKE_BINARY_DIR}/xiaotai-native-mqtt" OR "esp-idf/mqtt.c" IN_LIST adapter)
            message(FATAL_ERROR "Dormant MQTT product unexpectedly generated or registered native sources")
        endif()
        return()
    endif()
    if(NOT "esp-idf/mqtt.c" IN_LIST adapter OR NOT "common/mqtt_config.c" IN_LIST adapter)
        message(FATAL_ERROR "Missing native port implementation or shared MQTT rules")
    endif()
    if(mode STREQUAL "new")
        if(EXISTS "${CMAKE_BINARY_DIR}/xiaotai-native-mqtt" OR NOT actual STREQUAL absolute)
            message(FATAL_ERROR "Default MQTT graph must retain stock source files")
        endif()
        get_target_property(actual_outbox native_outbox SOURCES)
        get_target_property(actual_utils native_utils SOURCES)
        if(NOT actual_outbox STREQUAL "${MQTT_DIR}/lib/mqtt_outbox.c" OR
           NOT actual_utils STREQUAL "${MQTT_DIR}/lib/mqtt_utils/mqtt_utils.c")
            message(FATAL_ERROR "Default MQTT graph must retain stock outbox and utils")
        endif()
        return()
    endif()
    file(GLOB manifests "${CMAKE_BINARY_DIR}/xiaotai-native-mqtt/*/manifest.json")
    list(LENGTH manifests count)
    if(NOT count EQUAL 1)
        message(FATAL_ERROR "Expected one consistent generated native source tree")
    endif()
    list(GET manifests 0 manifest)
    get_filename_component(generated "${manifest}" DIRECTORY)
    foreach(pair IN ITEMS "native_mqtt;${main_sources}" "native_outbox;lib/mqtt_outbox.c"
                          "native_utils;lib/mqtt_utils/mqtt_utils.c")
        list(POP_FRONT pair target)
        get_target_property(actual "${target}" SOURCES)
        set(expected)
        foreach(source IN LISTS pair)
            list(APPEND expected "${generated}/${source}")
        endforeach()
        if(NOT actual STREQUAL expected)
            message(FATAL_ERROR "Mixed MQTT source graph: ${target}: ${actual}")
        endif()
        get_target_property(paths "${target}" INCLUDE_DIRECTORIES)
        foreach(path IN LISTS paths)
            if(path MATCHES "^${MQTT_DIR}/" OR NOT IS_DIRECTORY "${path}")
                message(FATAL_ERROR "Original or invalid native include survived: ${target}: ${path}")
            endif()
        endforeach()
    endforeach()
endfunction()
cmake_language(DEFER CALL assert_mqtt_source_graph)
